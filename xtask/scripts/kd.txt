# Roteiro do teste `kd` (cargo xtask test-boot --kd=com2 --script=xtask/scripts/kd.txt):
# o cliente do depurador do kernel anexa pela COM2, para a máquina, consulta
# o estado do OpenNT, lê e grava memória, pede a parada com a máquina
# rodando, usa breakpoints de software e de hardware num ponto do laço
# principal do kernel, consulta objetos do NT (processo, thread, driver e
# dispositivo), dá um passo, rastreia instruções passo a passo dentro do
# convidado e se desliga; a máquina segue até o fim do boot (os marcadores
# da COM1 são do comando do teste). Usa também a linguagem dos roteiros
# (docs/ARQUITETURA.md): o tratador de parada (on-break), as condições, os
# laços pelos processos, threads e módulos e as variáveis tiradas da saída.
# A transcrição da sessão fica em kd-sessao.log, ao lado do serial.log.

# Anexar: o primeiro pacote com a máquina rodando a para no tique seguinte.
attach
expect-stop SIGINT
kd
expect KdDebuggerEnabled=1 KdDebuggerNotPresent=0
cpulist
expect processadores parados: 1,2,3,4
cpus

# Memória: o cabeçalho do PE do kernel na VA em que ele roda, e uma escrita
# num byte sem uso do cabeçalho DOS, conferida e desfeita.
expect-eq word[0xfffff80040000000] 0x5a4d
save cabecalho dword[0xfffff80040000040]
poke 4 0xfffff80040000040 0xdeadbeef
expect-eq dword[0xfffff80040000040] 0xdeadbeef
poke 4 0xfffff80040000040 $cabecalho
expect-eq dword[0xfffff80040000040] $cabecalho

# Pedido de parada (0x03) com a máquina rodando.
continue
sleep 1500
break
expect-stop SIGINT

# Breakpoint de software numa função do laço principal do kernel, com uma
# linha de log a cada acerto (sai na transcrição na hora) e um tratador de
# parada: nos dois primeiros acertos o tratador imprime e a máquina segue
# sozinha; no terceiro ele a deixa parada e o wait-stop termina.
bp kernel::ntos::ke::run_kernel_tasks /w "1" /log "laco rsp={@rsp}"
on-break kernel::ntos::ke::run_kernel_tasks {
    print acerto {$hits:d} em {@rip:s}
    if $hits >= 3 {
        stop
    }
}
continue
wait-stop 600000
expect-stop swbreak
expect-eq %rip kernel::ntos::ke::run_kernel_tasks
expect-eq $hits 3
expect-console laco rsp=
on-break kernel::ntos::ke::run_kernel_tasks off

# Os registradores além do pacote `g`: a metade alta dos YMM sai do XSAVE na
# CPU da parada (o kernel liga o AVX no boot).
rx
expect ymm15h

# A saída do DbgPrint do ksecdd.sys real chegou durante o boot.
expect-console SspiInitAsyncInterface
log
expect SspiInitAsyncInterface

# Estado do OpenNT na parada.
ps
expect System
# O laço pelos processos acha o System.
for-each process sistema name System {
    leave
}
expect-eq $sistema_pid 4
threads
# As quatro primeiras threads de sistema depois da do boot são as worker
# threads do ExpWorkerInitialization, paradas à espera de itens (uma delas
# no FsRtlWaitForSmssEvent, à espera do smss): a pilha pelo quadro de troca
# chega à rotina da worker.
save-word worker Waiting 2
btt $worker
expect kernel::ntos::io::work_items::worker_thread
# Uma thread de kernel bloqueada: a primeira parada da lista com um quadro
# do acpi.sys real na pilha (uma thread de sistema dele, criada na árvore de
# dispositivos, esperando os eventos dela). Registradores e pilha pelo
# quadro de troca da pilha de kernel dela, até o código do driver.
threads
save-thread esperando Waiting 2 ACPI
# O mesmo pelo laço das threads: a primeira parada com um quadro do
# acpi.sys na pilha.
for-each thread acpi state Waiting stack ACPI {
    leave
}
expect-eq $acpi $esperando
context $esperando
expect quadro de troca
btt $esperando
expect kernel::ntos::ke::wait::
expect ACPI
lm
expect ksecdd.sys
for-each module ksecdd name ksecdd {
    leave
}
expect-eq $ksecdd_count 1
cpus
irql
obj \Device
expect tipo Directory
bt
expect kernel::ntos::init::main::kmain
k 4
expect Child-SP
? 1+2*3
expect = 0x7
expect-eq poi(@rsp) [%rsp]

# Objetos do NT pelas extensões do depurador: o processo System, a thread
# bloqueada, o objeto de driver do acpi.sys real e um dispositivo dele com
# a pilha de dispositivos.
!process 4
expect EPROCESS
expect threads:
save-after sistema_ep EPROCESS
expect-eq $sistema_ep $sistema
!thread $esperando
expect motivo da espera
!drvobj \Driver\ACPI
expect DRIVER_OBJECT
expect IRP_MJ_PNP
save-word dispositivo dispositivos: 1
!devobj $dispositivo
expect DEVICE_OBJECT
expect pilha de dispositivos

# Objetos do NT pelos tipos do PDB do ntoskrnl (EPROCESS, ETHREAD com os
# blocos de espera, HANDLE_TABLE, OBJECT_HEADER com o descritor) e as
# consultas do stub sobre o que vive só nas coleções do kernel (a tabela de
# tipos, o token, as portas de conclusão e as ALPC de um processo).
!obtypes
expect ObHeaderCookie
expect Process
!procs
expect EPROCESS
expect System
!threads 4
expect ETHREAD
expect Waiting
save-word bloqueada Waiting 0
!waits $bloqueada
expect BLOCO
expect WaitBlockActive
!handles 4
expect mostrados
# O OBJECT_HEADER de um objeto do ObCreateObject (o evento chaveado com
# nome da inicialização do executivo): o TypeIndex decodificado pela
# ObTypeIndexTable e as contagens. Os objetos que o OpenNT ainda cria sem
# o cabeçalho do Ob (processos, drivers, diretórios) dão um erro claro.
!obj \KernelObjects\CritSecOutOfMemoryEvent
expect tipo KeyedEvent
expect PointerCount
!token 4
expect TOKEN
expect usuario S-1-5-18
!iocp 4
expect handles IoCompletion
!alpc 4
expect handles ALPC_Port
lmv ksecdd
expect ksecdd.sys
# A base que o cliente usa nos símbolos é a da lista do alvo.
save-after base_ksecdd "  base "
expect-eq $base_ksecdd $ksecdd

# Ferramentas de engenharia reversa: o Registro vivo, o namespace do Object
# Manager e o monitor de APIs e o rastreador de IRPs, ligados e desligados
# com a maquina parada.
reg \Registry\Machine\SYSTEM\Select
expect Current
objdir \Driver
expect ACPI
apimon on
expect monitor ligado
irptrace on
expect rastreador ligado
apimon off
irptrace off
apimon status
expect desligado

# Um passo: o `push rbp` do prólogo.
save rsp0 %rsp
save rbp0 %rbp
step
expect-stop SIGTRAP
expect-eq %rsp $rsp0-8
expect-eq [%rsp] $rbp0
bc all

# Breakpoint de hardware (DR0) na mesma função.
hbp kernel::ntos::ke::run_kernel_tasks
bl
expect DR0
continue
wait-stop 60000
expect-stop hwbreak
expect-eq %rip kernel::ntos::ke::run_kernel_tasks

# Rastro de instruções passo a passo dentro do convidado, a partir da
# entrada da função: 300 instruções, a árvore de chamadas e o resto da
# função até o retorno.
trace 300
expect 300 instrucoes executadas
trace show 0 2
expect kernel::ntos::ke::run_kernel_tasks
trace tree 10
expect instrucoes
trace ret 400000
expect a funcao retornou
trace status
expect retorno da funcao
hbc kernel::ntos::ke::run_kernel_tasks

# Desliga: a máquina segue até o fim do boot (o shell e o serviço de
# teclado lendo do driver real, em qualquer ordem).
detach
wait-serial /t 300000 [shell] pronto
wait-serial [kbd] lendo teclas do driver real
