# Roteiro do teste `kd` (cargo xtask test-boot --kd=com2 --script=xtask/scripts/kd.txt):
# o cliente do depurador do kernel anexa pela COM2, para a máquina, consulta
# o estado do OpenNT, lê e grava memória, pede a parada com a máquina
# rodando, usa breakpoints de software e de hardware num ponto do laço
# principal do kernel, consulta objetos do NT (processo, thread, driver e
# dispositivo), dá um passo, rastreia instruções passo a passo dentro do
# convidado e se desliga; a máquina segue até o fim do boot (os marcadores
# da COM1 são do comando do teste).

# Anexar: o primeiro pacote com a máquina rodando a para no tique seguinte.
attach
expect-stop SIGINT
kd
expect KdDebuggerEnabled=1 KdDebuggerNotPresent=0
cpulist
expect processadores parados: 1,2,3,4
cpus

# Memória: o cabeçalho do PE do kernel na VA em que ele roda, e uma escrita
# num byte sem uso do cabeçalho DOS, conferida e desfeita.
expect-eq word[0xfffff80040000000] 0x5a4d
save cabecalho dword[0xfffff80040000040]
poke 4 0xfffff80040000040 0xdeadbeef
expect-eq dword[0xfffff80040000040] 0xdeadbeef
poke 4 0xfffff80040000040 $cabecalho
expect-eq dword[0xfffff80040000040] $cabecalho

# Pedido de parada (0x03) com a máquina rodando.
continue
sleep 1500
break
expect-stop SIGINT

# Breakpoint de software numa função do laço principal do kernel.
bp kernel::ntos::ke::run_kernel_tasks
continue
wait-stop 600000
expect-stop swbreak
expect-eq %rip kernel::ntos::ke::run_kernel_tasks

# A saída do DbgPrint do ksecdd.sys real chegou durante o boot.
expect-console SspiInitAsyncInterface
log
expect SspiInitAsyncInterface

# Estado do OpenNT na parada.
ps
expect System
threads
# Uma thread de kernel bloqueada (a primeira da lista é uma thread de
# sistema do acpi.sys real, criada na árvore de dispositivos, esperando os
# eventos dela): registradores e pilha pelo quadro de troca da pilha de
# kernel dela, até o código do driver.
save-word esperando Waiting 2
context $esperando
expect quadro de troca
btt $esperando
expect kernel::ntos::ke::wait::
expect ACPI
lm
expect ksecdd.sys
cpus
irql
obj \Device
expect tipo Directory
bt
expect kernel::ntos::init::main::kmain
k 4
expect Child-SP
? 1+2*3
expect = 0x7
expect-eq poi(@rsp) [%rsp]

# Objetos do NT pelas extensões do depurador: o processo System, a thread
# bloqueada, o objeto de driver do acpi.sys real e um dispositivo dele com
# a pilha de dispositivos.
!process 4
expect EPROCESS
expect threads:
!thread $esperando
expect motivo da espera
!drvobj \Driver\ACPI
expect DRIVER_OBJECT
expect IRP_MJ_PNP
save-word dispositivo dispositivos: 1
!devobj $dispositivo
expect DEVICE_OBJECT
expect pilha de dispositivos
lmv ksecdd
expect ksecdd.sys

# Extensões para os processos de usuário: o processo das leituras, a
# tabela de handles do System, as faixas do sistema, os pontos de evento e
# as exceções do anel 3.
.process 4
expect EPROCESS
.process 0
expect o do processador
handles 4
expect EPROCESS
kmap
expect pool do kernel
events
expect opennt_kd_image_event
uex
expect excecoes do anel 3

# Um passo: o `push rbp` do prólogo.
save rsp0 %rsp
save rbp0 %rbp
step
expect-stop SIGTRAP
expect-eq %rsp $rsp0-8
expect-eq [%rsp] $rbp0
bc all

# Breakpoint de hardware (DR0) na mesma função.
hbp kernel::ntos::ke::run_kernel_tasks
bl
expect DR0
continue
wait-stop 60000
expect-stop hwbreak
expect-eq %rip kernel::ntos::ke::run_kernel_tasks

# Rastro de instruções passo a passo dentro do convidado, a partir da
# entrada da função: 300 instruções, a árvore de chamadas e o resto da
# função até o retorno.
trace 300
expect 300 instrucoes executadas
trace show 0 2
expect kernel::ntos::ke::run_kernel_tasks
trace tree 10
expect instrucoes
trace ret 400000
expect a funcao retornou
trace status
expect retorno da funcao
hbc kernel::ntos::ke::run_kernel_tasks

# Desliga: a máquina segue até o fim do boot (o shell e o serviço de
# teclado lendo do driver real, em qualquer ordem).
detach
wait-serial [shell] pronto
wait-serial [kbd] lendo teclas do driver real
