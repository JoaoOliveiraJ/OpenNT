# Roteiro do teste `kd-usuario` (cargo xtask test-boot --kd=com2
# --initial-process=smss --script=xtask/scripts/kd-usuario.txt): um
# breakpoint de usuário num processo de verdade, o csrss.exe que o smss.exe
# real cria. O breakpoint é adiado (a ntdll ainda não está no processo
# quando ele é pedido), arma na carga da ntdll no csrss.exe e é de log: a
# cada acerto o cliente registra a linha e continua, e o stub sai do
# breakpoint por um passo da thread (o byte original por uma instrução, o
# TF e o 0xCC de volta no #DB). O #DB desse passo é do depurador: não pode
# chegar ao processo como STATUS_SINGLE_STEP (0x80000004), nem quando outra
# thread sai de um breakpoint antes de ele chegar, nem quando o host pede um
# passo (s) numa parada de usuário e continua depois. O csrss.exe segue vivo
# depois dos acertos e o boot chega ao fim do roteiro do smss. No fim, a
# pilha de cada thread em espera do smss.exe e do csrss.exe chega ao começo
# da thread, também a das threads sem serviço suspenso (uma espera adiada),
# lida do CONTEXT do anel 3 delas. Com o anel 3 nos APs (`user_smp` 1, o
# teste `kd-usuario-smp2`), uma parada de usuário forçada num AP, com o
# passo dela e os outros processadores congelados.

attach
expect-stop SIGINT

# Adiado, só no csrss.exe, com log, desligado depois de 200 acertos.
bu /p csrss.exe ntdll!RtlAllocateHeap /log "heap tamanho {@r8:d}" /max 200
expect modelo
# Outro adiado no csrss.exe que para uma vez: um passo da thread nessa
# parada e o continuar depois dele.
bu /p csrss.exe ntdll!RtlFreeHeap /max 1
expect modelo
continue
wait-stop 600000
expect-stop RtlFreeHeap
s
expect-stop SIGTRAP
expect-stop RtlFreeHeap+

# Com o anel 3 nos APs: um breakpoint no csrss.exe que só para no AP (a
# condição sobre `$cpu`, o processador da parada; os acertos em outra CPU
# seguem sem contar). A parada é do AP (o dono, KPRCB.IpiFrozen 4), com o
# BSP congelado (2), e o passo da thread também para.
if $user_smp == 1 {
    bu /p csrss.exe ntdll!RtlLeaveCriticalSection /w "$cpu == 1" /max 1
    continue
    wait-stop 600000
    expect-stop RtlLeaveCriticalSection
    expect-eq $cpu 1
    cpulist
    expect processadores parados: ${processadores}
    # A thread corrente do AP é a do csrss.exe.
    cpus
    expect csrss.exe
    monitor nmi
    expect CPU 1: dono da parada (KPRCB.IpiFrozen 4)
    expect CPU 0: congelado
    expect (KPRCB.IpiFrozen 2)
    s
    expect-stop SIGTRAP
    expect-stop RtlLeaveCriticalSection+
}
continue
wait-serial /t 600000 SMSS_CSRSS_OK
break
expect-stop SIGINT

# O breakpoint armou no csrss.exe e registrou acertos.
expect-console heap tamanho
bl
expect acertos
ps
expect csrss.exe
if serial-has "first chance 0x80000004" {
    fail o passo do depurador chegou ao processo como STATUS_SINGLE_STEP
}

# Sem o breakpoint, o csrss.exe continua rodando.
bc all
continue
sleep 3000
break
expect-stop SIGINT
ps
expect csrss.exe

# A pilha das threads em espera do smss.exe e do csrss.exe, parados há 3 s:
# a de uma espera adiada (as do porto ALPC do smss.exe) sai do CONTEXT do
# anel 3, sem quadro de troca; a de um serviço bloqueado sai do quadro de
# troca e atravessa a syscall. As duas vão até o começo da thread.
set anel3 nao
for-each thread t process smss state Waiting {
    btt $t 48
    expect-not #0  0x0000000000000000
    expect-not -- parou:
    expect -- fim da pilha
    if output-has "no anel 3" {
        set anel3 sim
    }
}
for-each thread t process csrss state Waiting {
    btt $t 48
    expect-not #0  0x0000000000000000
    expect-not -- parou:
    expect -- fim da pilha
    if output-has "no anel 3" {
        set anel3 sim
    }
}
if "${anel3}" != "sim" {
    fail nenhuma thread em espera adiada (a pilha pelo CONTEXT do anel 3)
}
if serial-has "first chance 0x80000004" {
    fail o passo do depurador chegou ao processo como STATUS_SINGLE_STEP
}
if serial-has "status=0x80000004" {
    fail uma thread saiu com STATUS_SINGLE_STEP
}

detach
